/* Converte valores antigos (texto livre, ex: "Speed", "MTB ") para o padrÃ£o */
export function normalizarTipo(tipo) {
    const t = String(tipo ?? '')
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .trim()
        .toLowerCase();

    if (['mtb', 'mountain', 'mountain bike'].includes(t)) return 'mtb';
    if (['road', 'speed', 'estrada', 'strada'].includes(t)) return 'road';
    if (t === 'gravel') return 'gravel';
    if (['urbana', 'urban', 'city', 'cidade'].includes(t)) return 'urbana';
    return 'outra';
}

import prisma from '../lib/prisma.js';
import modeloBikeService from './modeloBikeService.js';
/* Este arquivo contÃ©m os serviÃ§os relacionados Ã s bikes dos usuÃ¡rios */

async function getBikesByUserId(userId) {
    return await prisma.bike.findMany({
        where: { userId },
        include: { modeloBike: true, pecas: true }
    });
}

/*crio a bike reaproveitando ou cadastrando o modelo no catÃ¡logo, e jÃ¡ crio uma peÃ§a para cada componente do catÃ¡logo.
  pecas = componentes com manutenÃ§Ã£o recente informada no onboarding: { chave, data, kmInstalacao? }
  - componente informado sem km de instalaÃ§Ã£o: assume o km atual da bike
  - componente nÃ£o informado: conta o desgaste desde o km 0 da bike*/
async function createBike({ userId, apelido, ano, modelo, marca, tipo, kmAtual, pecas }) {
    const modeloBike = await modeloBikeService.findOrCreateModelo({ modelo, marca, tipo });
    const componentes = await prisma.componenteReferencia.findMany();

    const km = Number(kmAtual) || 0;
    const informadas = new Map((pecas ?? []).map((p) => [p.chave, p]));

    return await prisma.bike.create({
        data: {
            apelido,
            ano,
            kmAtual: km,
            userId,
            modeloBikeId: modeloBike.id,
            pecas: {
                create: componentes.map((c) => {
                    const info = informadas.get(c.chave);
                    const kmInstalacao = info
                        ? (info.kmInstalacao != null && info.kmInstalacao !== '' ? Number(info.kmInstalacao) : km)
                        : 0;
                    return {
                        nome: c.nome,
                        componenteId: c.id,
                        kmInstalacao,
                        ultimaManutencao: info ? new Date(info.data) : undefined,
                        criadaEm: new Date()
                    };
                })
            }
        },
        include: { modeloBike: true, pecas: true }
    });
}

export default {
    getBikesByUserId,
    createBike
};

import prisma from '../lib/prisma.js';
import pecaService from './pecaService.js';

async function createManutencao({ componenteChave, ...data }) {
    // Salva o histÃ³rico
    const manutencao = await prisma.historicoManutencao.create({
        data
    });

    // Atualiza a quilometragem atual da bike se a informada na manutenÃ§Ã£o for maior
    const bike = await prisma.bike.findUnique({ where: { id: data.bikeId } });
    if (bike && data.km > bike.kmAtual) {
        await prisma.bike.update({
            where: { id: data.bikeId },
            data: { kmAtual: data.km }
        });
    }

    // Em uma troca, a peÃ§a passa a contar o desgaste a partir deste km e desta data
    if (data.tipo === 'troca' && componenteChave) {
        await pecaService.registrarTroca({
            bikeId: data.bikeId,
            componenteChave,
            km: data.km,
            data: data.data
        });
    }

    return manutencao;
}

async function getManutencoesByBike(bikeId) {
    // Busca todas as manutenÃ§Ãµes da bike ordenadas da mais recente para a mais antiga
    return await prisma.historicoManutencao.findMany({
        where: { bikeId },
        orderBy: { data: 'desc' }
    });
}

export default {
    createManutencao,
    getManutencoesByBike
};

import prisma from '../lib/prisma.js';

/* Este arquivo contÃ©m os serviÃ§os relacionados ao catÃ¡logo de modelos de bike, compartilhado entre todos os usuÃ¡rios */

async function searchModelos(termo) {
    return await prisma.modeloBike.findMany({
        where: {
            modelo: { contains: termo, mode: 'insensitive' }
        },
        take: 10
    });
}

async function findModeloByNome(nome) {
    return await prisma.modeloBike.findFirst({
        where: {
            modelo: { equals: nome, mode: 'insensitive' }
        }
    });
}

async function findOrCreateModelo({ modelo, marca, tipo }) {
    const existente = await findModeloByNome(modelo);
    if (existente) return existente;

    if (!marca || !tipo) {
        throw new Error('marca e tipo sÃ£o obrigatÃ³rios para cadastrar um novo modelo');
    }
    if (!TIPOS_BIKE.includes(tipo)) {
        throw new Error(`tipo invÃ¡lido. Use um destes: ${TIPOS_BIKE.join(', ')}`);
    }

    return await prisma.modeloBike.create({
        data: { modelo, marca, tipo }
    });
}

export default {
    searchModelos,
    findModeloByNome,
    findOrCreateModelo
};

        prisma.componenteReferencia.findMany(),
        prisma.peca.findMany({ where: { bikeId } })
    ]);
    if (!bike) return;

    for (const peca of pecas.filter((p) => !p.componenteId)) {
        const componente = componentes.find((c) => c.chave === NOMES_LEGADOS[peca.nome]);
        if (!componente) continue;

        const data = { componenteId: componente.id, nome: componente.nome };
        // peÃ§a com manutenÃ§Ã£o informada e sem km de instalaÃ§Ã£o: assume que foi feita no km atual
        if (peca.ultimaManutencao && peca.kmInstalacao === 0) {
            data.kmInstalacao = bike.kmAtual;
        }
        await prisma.peca.update({ where: { id: peca.id }, data });
        peca.componenteId = componente.id;
    }

    const existentes = new Set(pecas.map((p) => p.componenteId).filter(Boolean));
    const faltando = componentes.filter((c) => !existentes.has(c.id));
    if (faltando.length > 0) {
        await prisma.peca.createMany({
            data: faltando.map((c) => ({
                nome: c.nome,
                bikeId,
                componenteId: c.id,
                kmInstalacao: 0,
                criadaEm: new Date()
            }))
        });
    }
}

/*escolho a faixa de vida Ãºtil conforme o tipo da bike: MTB e Road tÃªm faixa prÃ³pria, os demais usam a genÃ©rica*/
function limitesDoComponente(componente, tipo) {
    let min = componente.vidaMin;
    let max = componente.vidaMax;

    if (tipo === 'mtb' && componente.vidaMaxMtb != null) {
        min = componente.vidaMinMtb ?? min;
        max = componente.vidaMaxMtb;
    } else if (tipo === 'road' && componente.vidaMaxRoad != null) {
        min = componente.vidaMinRoad ?? min;
        max = componente.vidaMaxRoad;
    }

    if (max == null) return null;
    if (min == null) min = max * 0.75;
    return { min, max };
}

function mesesDesde(data) {
    return (Date.now() - new Date(data).getTime()) / (1000 * 60 * 60 * 24 * 30.4375);
}

/*bom: abaixo de min | medio: entre min e max | ruim: a partir de max*/
function calcularStatus(peca, bike, componente) {
    const tipo = normalizarTipo(bike.modeloBike?.tipo);
    const limites = limitesDoComponente(componente, tipo);

    let desgaste;
    if (componente.unidade === 'meses') {
        const base = peca.ultimaManutencao ?? peca.criadaEm;
        desgaste = base ? mesesDesde(base) : 0;
    } else {
        desgaste = Math.max(0, bike.kmAtual - peca.kmInstalacao);
    }

    let status = 'bom';
    let vidaRestante = 100;
    if (limites) {
        vidaRestante = Math.max(0, Math.round(100 - (desgaste / limites.max) * 100));
        if (desgaste >= limites.max) status = 'ruim';
        else if (desgaste >= limites.min) status = 'medio';
    }

    return {
        id: peca.id,
        chave: componente.chave,
        nome: componente.nome,
        unidade: componente.unidade,
        desgaste: Math.round(desgaste * 10) / 10,
        limiteMin: limites?.min ?? null,
        limiteMax: limites?.max ?? null,
        vidaRestante,
        status,
        sinaisDesgaste: componente.sinaisDesgaste,
        observacao: componente.observacao ?? null,
        fonte: componente.fonte
    };
}

async function getStatusByBike(bikeId) {
    await sincronizarPecas(bikeId);

    const bike = await prisma.bike.findUnique({
        where: { id: bikeId },
        include: { modeloBike: true }
    });
    if (!bike) return null;

    const pecas = await prisma.peca.findMany({
        where: { bikeId, componenteId: { not: null } },
        include: { componente: true }
    });

    const status = pecas
        .filter((p) => p.componente)
        .sort((a, b) => a.componente.ordem - b.componente.ordem)
        .map((p) => calcularStatus(p, bike, p.componente));

    return {
        bike: {
            id: bike.id,
            apelido: bike.apelido,
            kmAtual: bike.kmAtual,
            tipo: normalizarTipo(bike.modeloBike?.tipo)
        },
        pecas: status
    };
}

/*registro a troca: a peÃ§a passa a contar o desgaste a partir do km e da data informados*/
async function registrarTroca({ bikeId, componenteChave, km, data }) {
    await sincronizarPecas(bikeId);

    const componente = await prisma.componenteReferencia.findUnique({ where: { chave: componenteChave } });
    if (!componente) return null;

    const peca = await prisma.peca.findFirst({ where: { bikeId, componenteId: componente.id } });
    if (!peca) return null;

    return await prisma.peca.update({
        where: { id: peca.id },
        data: { kmInstalacao: km, ultimaManutencao: data }
    });
}

export default function Login() {
  const router = useRouter();
  const [email, setEmail] = useState('');
  const [senha, setSenha] = useState('');

  async function handleLogin() {
    if (!email || !senha) {
      alert('Preencha todos os campos');
      return;
    }
    
    try {
      // 1. Faz a requisiÃ§Ã£o para a API
      const response = await axios.post(`${API_URL}/login`, {
        email,
        password: senha,
      });
      
      const usuario = response.data;
      
      
      await AsyncStorage.setItem('@user_id', String(usuario.id));

      if (!usuario.bikes || usuario.bikes.length === 0) {
        router.replace({ pathname: '/onboarding', params: { userId: usuario.id } });
      } else {
        await AsyncStorage.setItem('@bike_id', String(usuario.bikes[0].id));
        router.replace('/(tabs)');
      }
            
    } catch (error: any) {
      console.log(error?.message);
      alert('Email ou senha invÃ¡lidos');
    }
  }

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Login</Text>
      <TextInput style={styles.input} placeholder="Email" value={email} onChangeText={setEmail} />

      <TouchableOpacity style={styles.button} onPress={handleLogin}>
        <Text style={styles.buttonText}>Entrar</Text>
      </TouchableOpacity>
      <TouchableOpacity onPress={() => router.push('/cadastro')}>

export default function Onboarding() {
  const router = useRouter();

  const { userId, origem } = useLocalSearchParams<{ userId: string; origem?: string }>();

  const isNovaBike = origem === 'minhas-bikes';

  const [marca, setMarca] = useState('');

  const [kmAtual, setKmAtual] = useState('');

  const [componentes, setComponentes] = useState<Componente[]>([]);
  const [pecasSelecionadas, setPecasSelecionadas] = useState<string[]>([]); // guarda a "chave" do componente

  const [enviando, setEnviando] = useState(false);

        setSugestoes(response.data);

        (m: ModeloSugestao) => m.modelo.toLowerCase() === modelo.trim().toLowerCase()

        selecionarSugestao(encontrado);

        setEtapa('marca-tipo');

      pecas.push({

import { useEffect, useState } from 'react';
import { useColorScheme as useRNColorScheme } from 'react-native';

/**
 * To support static rendering, this value needs to be re-calculated on the client side for web
 */
export function useColorScheme() {
  const [hasHydrated, setHasHydrated] = useState(false);

  useEffect(() => {
    setHasHydrated(true);
  }, []);

  const colorScheme = useRNColorScheme();

  if (hasHydrated) {
    return colorScheme;
  }

  return 'light';
}

import axios from 'axios';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { API_URL } from '@/constants/api';

/* Retorna o id da bike ativa. Se ainda nÃ£o estiver salvo, busca a primeira bike do usuÃ¡rio na API. */
export async function getBikeId(): Promise<string | null> {
  const salvo = await AsyncStorage.getItem('@bike_id');
  if (salvo) return salvo;

  const userId = await AsyncStorage.getItem('@user_id');
  if (!userId) return null;

  const { data } = await axios.get(`${API_URL}/bikes/usuario/${userId}`);
  if (data.length > 0) {
    await AsyncStorage.setItem('@bike_id', String(data[0].id));
    return String(data[0].id);
  }
  return null;
}
